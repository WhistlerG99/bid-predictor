# Building the Training Image

# 1) Set environment variables
export AWS_REGION=us-east-1
export ACCOUNT_ID=$(aws sts get-caller-identity --query Account --output text) # it's usually 622055002283
export REPO=bid-predictor-sklearn-gpu
export TAG=latest

# 2) Create ECR repo (if not already there)
aws ecr create-repository --repository-name $REPO >/dev/null 2>&1 || true

# 3) Login to ECR
aws ecr get-login-password --region $AWS_REGION \
| docker login --username AWS --password-stdin \
  $ACCOUNT_ID.dkr.ecr.$AWS_REGION.amazonaws.com

# 4) Build and push the Image
docker buildx build \
  --platform linux/amd64 \
  --provenance=false --sbom=false \
  -f containers/Dockerfile.gpu \
  -t $ACCOUNT_ID.dkr.ecr.$AWS_REGION.amazonaws.com/$REPO:$TAG \
  --push .
